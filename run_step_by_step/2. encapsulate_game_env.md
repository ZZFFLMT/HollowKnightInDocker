## 安装jupyter-lab编写代码

```bash
# 安装jupyter lab及各种插件
## jupyterlab-language-pack-zh-CN 中文语言包
pip install jupyterlab jupyterlab-language-pack-zh-CN

# 生成配置文件, 配置文件默认存放在~/.jupyter/jupyter_server_config.py下
jupyter server --generate-config

# 向配置文件末尾添加以下配置
cat <<EOF > ~/.jupyter/jupyter_server_config.py
c.ServerApp.ip = '*'          # 允许登录jupyter的ip范围
c.ServerApp.port = 3389       # jupyter监听的端口
c.ServerApp.browser = 'False' # 是否自动打开浏览器
c.ServerApp.password = ''     # 设置jupyter的密码, 不起作用
c.ServerApp.root_dir = ''     # 指定jupyter的工作目录, 
                              # 若不写绝对路径的话, 会将其与当前路径拼接
EOF

# 自动填入密码
## 生成加密后的哈希密码
PASSWORD_HASH=$(python3 -c "from jupyter_server.auth import passwd; print(passwd('zzfflmt'))")
## 将哈希值写入 json 配置文件
cat <<EOF > ~/.jupyter/jupyter_server_config.json
{
  "IdentityProvider": {
    "hashed_password": "$PASSWORD_HASH"
  }
}
EOF

# 安装tmux来运行jupyter
apt update && apt install tmux -y
# 开启tmux终端
tmux
# 运行jupyter
cd /workspace
jupyter-lab --allow-root
```

```python
#!/usr/bin/env python3
# /workspace/STORM-main/hollow_knight_env/udp_listener_test.py
# 接受udp并打印的测试
import os, socket, json, time

HOST = os.getenv("HKRL_UDP_HOST", "0.0.0.0")   # __________________ 127.0.0.1
PORT = int(os.getenv("HKRL_UDP_PORT", "28115"))

sock = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
sock.bind((HOST, PORT))
print(f"[UDP] listening on {HOST}:{PORT}")

while True:
    data, addr = sock.recvfrom(65535)
    ts = time.ctime()
    print(f"[{ts}] from {addr[0]}:{addr[1]} {len(data)}B")
    try:
        obj = json.loads(data.decode("utf-8"))
        # __ JSON ____________________
        print(json.dumps(obj, ensure_ascii=False))
    except Exception:             
        # __ JSON ______________
        try:
            print(data.decode("utf-8", errors="replace"))
        except Exception:
            print(repr(data))
```
