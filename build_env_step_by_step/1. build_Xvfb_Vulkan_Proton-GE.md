# 安装Xvfb, Vulkan与Proton, 在显卡中运行空洞骑士

## 以docker compose方式启动容器

```yaml
services:
  show-lab:
    image: pytorch/pytorch:2.2.0-cuda12.1-cudnn8-devel
    container_name: show-gpu-lab
    hostname: zzfflmt
    restart: unless-stopped
    tty: true                 # 交互 TTY
    stdin_open: true
    init: true                # 更好地回收子进程
    shm_size: "4g"            # 足够的 /dev/shm (浏览器/图形工具常用)
    gpus: all                 # 需要 docker compose v2+ 支持
    environment:
      - NVIDIA_VISIBLE_DEVICES=all
      # 向容器内挂载驱动: CUDA/计算能力 管理与查询 硬件视频编码解码 3D渲染 显示与输出 32位兼容库
      - NVIDIA_DRIVER_CAPABILITIES=compute,utility,video,graphics,display,compat32
      - VK_ICD_FILENAMES=/usr/share/vulkan/icd.d/nvidia_icd.json
      # 下面三项只是占位, 后续如果你要接入宿主 X11/Wayland 再在容器内设置即可
      - DISPLAY=:577
      # - WAYLAND_DISPLAY
      # - XDG_RUNTIME_DIR=/tmp
      - STEAM_COMPAT_CLIENT_INSTALL_PATH=/tmp/steam-empty
      # - STEAM_COMPAT_DATA_PATH=/workspace/compatdata
    ports:
      # - "5900:5900"           # VNC(x11vnc/TigerVNC 等)使用, 可不开放不向外界暴露
      # - "5901:5901"           # 同上
      # - "3389:3389"           # RDP(若用 weston 的 RDP backend)
      # - "3390:3390"           # 同上
      - "6080:6080"           # noVNC(websockify 一般用这个)
    volumes:
      - /etc/localtime:/etc/localtime:ro # 映射时区, 避免下载部分软件需要设置时区
      - ./workspace:/workspace           # 持久化数据/前缀等(可改路径)
      # 以下内容不是必须, 这是为了使用apt的缓存设置的, 若没有建立简单的aptcache请进行注释
      #- ./container-data/apt-archives:/var/cache/apt/archives
      #- ./container-data/apt-lists:/var/lib/apt/lists
      #- ./container-config/apt/01proxy:/etc/apt/apt.conf.d/01proxy:ro
      #- ./container-config/apt/99-nvidia-block.pref:/etc/apt/preferences.d/99-nvidia-block:ro
    # devices:
      # - "/dev/snd:/dev/snd" # 音频设备
      # - "/dev/dri:/dev/dri" # 图形与渲染DRM子系统, 若使用Xvfb这块则无用
    security_opt:
      - seccomp=unconfined # 放宽安全限制, 允许容器内进程调用所有系统调用
```

## 运行虚拟桌面并运行NoVNC在浏览器中观察虚拟屏幕

```bash
apt update
# 安装Xvfb + x11vnc + novnc, xterm与openbox提供最基本的图形环境
apt-get install -y --no-install-recommends xvfb x11-apps x11-utils xfonts-base procps x11vnc novnc websockify openbox xterm
# 检查是否已经安装好
## 检查Xvfb是否存在
which Xvfb
## 检查x11vnc的版本信息
x11vnc --version
## 检查novnc的文件夹下是否有launch.sh
ls -l /usr/share/novnc/utils/

# 设定显示的屏幕
export DISPLAY=:577
# 使用Xvfb在该号上创建屏幕
Xvfb $DISPLAY -screen 0 1280x960x24 -ac -nolisten tcp >/tmp/xvfb.log 2>&1 &
# 检查
sleep 1
## 查看能否获取到屏幕信息
xdpyinfo | grep -E 'name of display|dimensions|depth of root' || true

# 启动openbox与拥有简单窗口管理器
openbox >/tmp/openbox.log 2>&1 &
# 开一个终端验证openbox是否可用
xterm >/tmp/xterm.log 2>&1 &
# 防止Wayland环境变量干扰
unset WAYLAND_DISPLAY XDG_SESSION_TYPE
# 挂载简单x11vnc
x11vnc -display $DISPLAY -rfbport 5900 -forever -shared -nopw -noxdamage  -o /tmp/x11vnc.log >/dev/null 2>&1 &
# 检查
## 查看是否启动正常
sleep 1
tail -n 5 /tmp/x11vnc.log
# 启动novnc将vnc请求转为websockify请求
/usr/share/novnc/utils/launch.sh --vnc localhost:5900 --listen 6080 >/tmp/novnc.log 2>&1 &

# 可选
## 安装x11的配置项
apt install -y x11-xserver-utils
## 调整桌面背景颜色
xsetroot -solid '#555555' || true
```


## 安装Vulkan库, 使用GPU进行渲染, 并运行空洞骑士

```bash
# 下载Vulkan相关的组件 相关库 mesa实现(其实只是icd)
apt-get install -y --no-install-recommends vulkan-tools libvulkan1 mesa-vulkan-drivers

## 手写NVIDIA的icd文件, 使其指向libGLX_nvidia.so.0
echo '{"file_format_version":"1.0.0","ICD":{"library_path":"libGLX_nvidia.so.0","api_version":"1.3"}}' > /usr/share/vulkan/icd.d/nvidia_icd.json
## 设置环境变量
export VK_ICD_FILENAMES=/usr/share/vulkan/icd.d/nvidia_icd.json 
# 测试(可选)
## 测试能否看到GPU名称
vulkaninfo | grep -E "GPU|Device Name|driverName|apiVersion" | head
## 测试是否正在调用GPU, 运行后终端会输出运行的设备
vkcube

# 安装GE-Proton10-27
## 解压安装包
tar zxvf /workspace/GE-Proton10-27.tar.gz -C /opt/
## 相当于WINEPREFIX
mkdir -p /tmp/steam-empty
export STEAM_COMPAT_CLIENT_INSTALL_PATH=/tmp/steam-empty
# 游戏目录及C盘位置
export STEAM_COMPAT_DATA_PATH=/workspace/compatdata/Hollow.Knight.GOG
# 屏幕左上角显示设备信息和fps
export DXVK_HUD=devinfo,fps
# 写入uuid, 防止wine报错
python3 -c 'import uuid; print(uuid.uuid4().hex)' > /etc/machine-id

# 安装32位的库(WOW64可能要用)
dpkg --add-architecture i386
apt-get update
apt-get install -y --no-install-recommends libstdc++6:i386 libgcc-s1:i386 zlib1g:i386 libx11-6:i386 libxext6:i386 libxrandr2:i386 libxrender1:i386 libxi6:i386 libxfixes3:i386 libasound2:i386 libpulse0:i386 libfreetype6:i386 libfontconfig1:i386 libc6:i386

# 先跑个winecfg, 设置虚拟桌面
/opt/GE-Proton10-27/proton run winecfg

# 再运行游戏
/opt/GE-Proton10-27/proton run explorer /desktop=HK1,1280x960 /workspace/compatdata/Hollow.Knight.GOG/pfx/drive_c/GOG\ Games/Hollow\ Knight/Hollow\ Knight.exe
```
